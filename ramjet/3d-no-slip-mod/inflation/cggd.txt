SetFactory("OpenCASCADE");

//======================================================
// 3D HALF NOZZLE + EXTENSION
// TOTAL LENGTH = 2.0411 m
// EXTRUSION WIDTH = 0.36 m
// STRUCTURED HEXA/QUAD-DOMINANT MESH
// TOPWALL REFINEMENT PRESERVED
//======================================================

//======================================================
// PARAMETERS
//======================================================

Lcurve    = 1.0;
Lstraight = 1.0;
Lexit     = 0.0411;

Hin   = 0.36;
Hmid  = 0.18;
Hexit = 0.1764;

// 3D width
W = 0.36;

//======================================================
// MESH PARAMETERS
//======================================================

Ncurve    = 100;
Nstraight = 100;
Nexit     = 20;

Nnormal   = 70;

// Fine layers at TOPWALL
BLratio = 0.892857;

// Number of layers through 3D width
Nwidth = 30;

//======================================================
// BOTTOM WALL
//======================================================

Point(1) = {0,0,0};
Point(2) = {1,0,0};
Point(3) = {2,0,0};
Point(4) = {2.0411,0,0};

Line(1) = {1,2};
Line(2) = {2,3};
Line(3) = {3,4};

//======================================================
// TOP WALL : 0 -> 1 m
// y = 0.36*x^3 - 0.54*x^2 + 0.36
//======================================================

p0 = newp;
Point(p0) = {0,Hin,0};

pts[] = {p0};

For i In {1:Ncurve}

    x = i/Ncurve;
    y = 0.36*x^3 - 0.54*x^2 + 0.36;

    p = newp;
    Point(p) = {x,y,0};

    pts[] += {p};

EndFor

Spline(4) = pts[];

//======================================================
// TOP WALL : 1 -> 2 m
//======================================================

Point(1001) = {2,Hmid,0};
Line(5) = {pts[#pts[]-1],1001};

//======================================================
// TOP WALL : 2 -> 2.0411 m
//======================================================

Point(1002) = {2.0411,Hexit,0};
Line(6) = {1001,1002};

//======================================================
// VERTICAL BOUNDARIES
//======================================================

Line(7) = {1,p0};

// x = 1 m
Line(9) = {2,pts[#pts[]-1]};

// x = 2 m
Line(10) = {3,1001};

// outlet
Line(8) = {4,1002};

//======================================================
// SURFACE 1 : 0 -> 1 m
//======================================================

Curve Loop(1) = {7,4,-9,-1};
Plane Surface(1) = {1};

//======================================================
// SURFACE 2 : 1 -> 2 m
//======================================================

Curve Loop(2) = {9,5,-10,-2};
Plane Surface(2) = {2};

//======================================================
// SURFACE 3 : 2 -> 2.0411 m
//======================================================

Curve Loop(3) = {10,6,-8,-3};
Plane Surface(3) = {3};

//======================================================
// STREAMWISE TRANSFINITE
//======================================================

Transfinite Curve {1,4}
    = Ncurve + 1 Using Progression 1;

Transfinite Curve {2,5}
    = Nstraight + 1 Using Progression 1;

Transfinite Curve {3,6}
    = Nexit + 1 Using Progression 1;

//======================================================
// WALL-NORMAL TRANSFINITE
// FINE CELLS AT TOPWALL
//======================================================

Transfinite Curve {7,9,10,8}
    = Nnormal + 1 Using Progression BLratio;

//======================================================
// TRANSFINITE SURFACES
//======================================================

Transfinite Surface {1}
    = {1,p0,pts[#pts[]-1],2};

Transfinite Surface {2}
    = {2,pts[#pts[]-1],1001,3};

Transfinite Surface {3}
    = {3,1001,1002,4};

//======================================================
// RECOMBINE 2D SURFACES
//======================================================

Recombine Surface {1,2,3};

//======================================================
// EXTRUDE TO 3D
//======================================================

Extrude {0,0,W}
{
    Surface{1,2,3};
    Layers{Nwidth};
    Recombine;
}

//======================================================
// PHYSICAL BOUNDARIES
//======================================================

Physical Surface("TOPWALL") = {3,4,10};

Physical Surface("BOTWALL") = {7,6,12};

Physical Surface("INLET") = {8};

Physical Surface("OUTLET") = {11};

Physical Surface("INTERFACE_1") = {9};

Physical Surface("INTERFACE_2") = {13};

Physical Volume("FLUID") = {1,2,3};

//======================================================
// MESH OPTIONS
//======================================================

Mesh.RecombineAll = 1;
Mesh.Smoothing = 10;
